-- [[ LÓGICA ESP ]]
local lines = {}
local function createLine()
    local l = Drawing.new("Line")
    l.Visible = false
    l.Color = Color3.fromRGB(255, 255, 255)
    l.Thickness = 1
    return l
end

local function updateESP()
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            if not lines[p] then lines[p] = createLine() end
            local line = lines[p]
            local hrp = p.Character.HumanoidRootPart
            local hum = p.Character:FindFirstChild("Humanoid")
            if espEnabled and hum and hum.Health > 0 then
                local vector, onScreen = Camera:WorldToViewportPoint(hrp.Position)
                if onScreen then
                    line.From = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y)
                    line.To = Vector2.new(vector.X, vector.Y)
                    line.Visible = true
                else line.Visible = false end
            else line.Visible = false end
        elseif lines[p] then lines[p].Visible = false end
    end
end

-- [[ LÓGICA AIMBOT COM FOV & VISIBILIDADE ]]
local function isVisible(part)
    local char = LocalPlayer.Character
    if not char then return false end
    local origin = Camera.CFrame.Position
    local direction = (part.Position - origin).Unit * (part.Position - origin).Magnitude
    local params = RaycastParams.new()
    params.FilterDescendantsInstances = {char, Camera}
    params.FilterType = Enum.RaycastFilterType.Exclude
    local result = workspace:Raycast(origin, direction, params)
    return result and result.Instance:IsDescendantOf(part.Parent)
end

local function getClosestInFOV()
    local target = nil
    local shortestDist = math.huge
    local center = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("Head") then
            local head = p.Character.Head
            local hum = p.Character:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 and isVisible(head) then
                local vector, onScreen = Camera:WorldToViewportPoint(head.Position)
                if onScreen then
                    local dist = (Vector2.new(vector.X, vector.Y) - center).Magnitude
                    if dist <= fovCircle.Radius and dist < shortestDist then
                        target = p
                        shortestDist = dist
                    end
                end
            end
        end
    end
    return target
end

-- [[ LOOP PRINCIPAL ]]
RunService.RenderStepped:Connect(function()
    updateESP()
    fovCircle.Position = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    fovCircle.Visible = showFOV
    
    if aimbotEnabled then
        local target = getClosestInFOV()
        if target and target.Character and target.Character:FindFirstChild("Head") then
            Camera.CFrame = CFrame.new(Camera.CFrame.Position, target.Character.Head.Position)
        end
    end
end)

Players.PlayerRemoving:Connect(function(p) if lines[p] then lines[p]:Remove() lines[p] = nil end end)
